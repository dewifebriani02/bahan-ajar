Indexing dengan loc dan iloc
⏱ ~10 menit

Sebelum menyaring data, kita harus paham cara mengakses sel, baris, atau kolom tertentu. Latih perbedaan loc (berdasarkan label) dan iloc (berdasarkan posisi).

Latihan loc — Sel 3

Python · Cell 3 — loc berbagai pola
# 3a. Satu nilai spesifik
print(df.loc[0, 'nama_bprs'])           # → BPRS Al-Amin

# 3b. Satu baris, beberapa kolom
print(df.loc[10, ['nama_bprs', 'kota', 'npf_persen']])

# 3c. Range baris (INKLUSIF) — perhatikan: hasilnya 4 baris bukan 3!
print(df.loc[0:3, 'nama_bprs'])

# 3d. Semua baris, hanya kolom finansial
kolom_fin = ['nama_bprs', 'npf_persen', 'roe_persen', 'car_persen']
df.loc[:, kolom_fin].head()

Latihan iloc — Sel 4

Python · Cell 4 — iloc berbagai pola
# 4a. Perbandingan langsung loc vs iloc
print("loc[0:3]  →", len(df.loc[0:3]), "baris")   # 4 baris (inklusif)
print("iloc[0:3] →", len(df.iloc[0:3]), "baris")  # 3 baris (eksklusif)

# 4b. 5 baris TERAKHIR, 3 kolom pertama
df.iloc[-5:, :3]

df.iloc[0::2][['nama_bprs', 'kota']]

Filter Kondisi Majemuk — Sel 6

Python · Cell 6 — kondisi AND, OR, NOT
# BPRS "ideal": NPF rendah DAN ROE tinggi — kandidat benchmark terbaik
df_ideal = df[(df['npf_persen'] < 2.5) & (df['roe_persen'] > 12)]
print("BPRS dengan NPF < 2.5% DAN ROE > 12%:")
print(df_ideal[['nama_bprs', 'npf_persen', 'roe_persen', 'total_aset_juta']])

# Setara dengan .query() — lebih bersih untuk kondisi kompleks
df_ideal_q = df.query('npf_persen < 2.5 and roe_persen > 12')
print(f"\nSama hasilnya? {len(df_ideal) == len(df_ideal_q)}")  # True

# BPRS bukan dari Jawa tapi aset di atas rata-rata — outlier positif?
rata_aset = df['total_aset_juta'].mean()
df_luar_jawa_besar = df[
    ~df['provinsi'].isin(jawa) &
    (df['total_aset_juta'] > rata_aset)
]
print(f"\nBPRS luar Jawa dengan aset di atas rata-rata ({rata_aset:,.0f} juta):")
print(df_luar_jawa_besar[['nama_bprs', 'provinsi', 'total_aset_juta']])

.transform() — Analisis Komparatif Peer — Sel 9

Python · Cell 9 — .transform() + kolom baru
# Hitung deviasi NPF setiap BPRS dari rata-rata provinsinya
df['rata_npf_prov'] = df.groupby('provinsi')['npf_persen'].transform('mean')
df['npf_vs_peer']   = (df['npf_persen'] - df['rata_npf_prov']).round(2)

# BPRS yang lebih baik dari peer-nya (deviasi negatif = NPF lebih rendah)
df_outperform = df[df['npf_vs_peer'] < 0].sort_values('npf_vs_peer')
print(df_outperform[['nama_bprs', 'provinsi', 'npf_persen', 'rata_npf_prov', 'npf_vs_peer']])

# Tambah juga: rank NPF dalam provinsi masing-masing
df['rank_npf_prov'] = df.groupby('provinsi')['npf_persen'].rank(method='min')
print("\nBPRS dengan rank terbaik (NPF terendah) di provinsinya:")
print(df[df['rank_npf_prov'] == 1][['nama_bprs', 'provinsi', 'npf_persen']])

Left Join — Sel 11

Python · Cell 11 — merge left join + apply
# Left join: pertahankan semua 20 BPRS
df_merged = pd.merge(df, benchmark_ojk, on='provinsi', how='left')
print(f"Shape setelah merge: {df_merged.shape}")  # tetap (20, N)
print(f"NaN di benchmark_npf: {df_merged['benchmark_npf'].isna().sum()}")

# Tambah kolom status vs benchmark menggunakan apply
def status_npf(row):
    if pd.isna(row['benchmark_npf']):
        return '📊 Tidak Ada Benchmark'
    elif row['npf_persen'] < row['benchmark_npf'] * 0.8:
        return '🌟 Jauh Di Bawah Benchmark'
    elif row['npf_persen'] <= row['benchmark_npf']:
        return '✅ Di Bawah Benchmark'
    else:
        return '⚠️ Melebihi Benchmark'

df_merged['status_npf'] = df_merged.apply(status_npf, axis=1)

# Tampilkan ringkasan
print("\nDistribusi status NPF vs Benchmark OJK:")
print(df_merged['status_npf'].value_counts())

# Detail BPRS yang melebihi benchmark
print("\nBPRS yang melebihi benchmark NPF OJK:")
mask = df_merged['status_npf'] == '⚠️ Melebihi Benchmark'
print(df_merged[mask][['nama_bprs', 'provinsi', 'npf_persen', 'benchmark_npf', 'kanwil_ojk']])

# Klasifikasi NPF → kategori kesehatan
df['kategori_npf'] = df['npf_persen'].apply(
    lambda x: '🌟 Sangat Sehat' if x < 2.0 else
              '✅ Sehat'         if x < 3.0 else
              '🟡 Cukup Sehat'   if x < 4.0 else
              '⚠️ Perlu Perhatian'

# Hitung usia dan klasifikasi era berdiri
